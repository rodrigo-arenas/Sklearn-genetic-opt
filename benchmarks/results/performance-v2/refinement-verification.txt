import json
import numpy as np
from sklearn.datasets import load_wine
from sklearn.tree import DecisionTreeClassifier
from threadpoolctl import threadpool_limits
from sklearn_genetic import GASearchCV, GAFeatureSelectionCV
from sklearn_genetic.space import Integer, Continuous

X,y=load_wine(return_X_y=True)
with threadpool_limits(limits=1):
    for kind in ('hpo','features'):
        results=[]
        for n_jobs in (1,2):
            kwargs=dict(population_size=4,generations=1,keep_top_k=2,local_search=True,
                local_search_top_k=2,local_search_steps=3,n_jobs=n_jobs,
                parallel_backend='population',random_state=15,verbose=False,cv=3)
            model=DecisionTreeClassifier(random_state=15,max_depth=4)
            search=(GASearchCV(model,param_grid={'max_depth':Integer(1,8),'ccp_alpha':Continuous(0.,.01)},**kwargs)
                    if kind=='hpo' else GAFeatureSelectionCV(model,max_features=8,**kwargs))
            search.fit(X,y)
            score=float(max(search.cv_results_['mean_test_score']))
            assert search.fit_stats_['local_refinement_candidates']==6
            assert search.fit_stats_['cross_validate_calls']<=18
            assert search.history['fitness_best'][-1]==score
            assert search.predict(X).shape==y.shape
            if kind=='features':
                assert 1<=search.support_.sum()<=8
            result={'kind':kind,'n_jobs':n_jobs,'cv_score':score,'fit_stats':search.fit_stats_,
                    'selected':search.best_params_ if kind=='hpo' else search.support_.tolist()}
            results.append(result)
            print(json.dumps(result),flush=True)
        assert results[0]['cv_score']==results[1]['cv_score']
        assert results[0]['selected']==results[1]['selected']
